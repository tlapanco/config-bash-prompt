#!/bin/bash # ============================================================ # CUSTOM BASH PROMPT # Personalización de la terminal Bash # ============================================================ # ------------------------------------------------------------ # Colores ANSI para los mensajes de salida # ------------------------------------------------------------ PURPLE='\033[1;35m' BLUE='\033[1;34m' GREEN='\033[1;32m' YELLOW='\033[1;33m' RED='\033[1;31m' CYAN='\033[1;36m' GRAY='\033[0;90m' BOLD='\033[1m' RESET='\033[0m' # ------------------------------------------------------------ # Configuración de archivos # ------------------------------------------------------------ # Ruta del archivo de configuración Bash del usuario BASHRC="$HOME/.bashrc" # Identificador para detectar si el prompt ya fue instalado MARKER="CUSTOM_BASH_PROMPT" # ------------------------------------------------------------ # Encabezado del instalador # ------------------------------------------------------------ echo "" echo -e "${PURPLE}${BOLD}╭──────────────────────────────────────────╮${RESET}" echo -e "${PURPLE}${BOLD}│ CUSTOM BASH PROMPT INSTALLER │${RESET}" echo -e "${PURPLE}${BOLD}╰──────────────────────────────────────────╯${RESET}" echo "" # ------------------------------------------------------------ # Verificar que exista el archivo .bashrc # ------------------------------------------------------------ if [[ ! -f "$BASHRC" ]]; then echo -e "${RED}[ERROR]${RESET} No se encontró el archivo .bashrc" echo -e "${YELLOW}[INFO]${RESET} Creando archivo de configuración..." touch "$BASHRC" || { echo -e "${RED}[ERROR]${RESET} No se pudo crear .bashrc" exit 1 } fi # ------------------------------------------------------------ # Comprobar si el prompt ya está instalado # ------------------------------------------------------------ if grep -q "$MARKER" "$BASHRC"; then echo -e "${YELLOW}[!]${RESET} El prompt personalizado ya está instalado." echo -e "${GRAY}[INFO]${RESET} No se realizaron modificaciones." else # -------------------------------------------------------- # Agregar configuración del prompt personalizado # -------------------------------------------------------- echo -e "${CYAN}[*]${RESET} Instalando prompt personalizado..." cat <<'EOF' >> "$BASHRC" # ============================================================ # CUSTOM_BASH_PROMPT # Prompt personalizado con colores ANSI # ============================================================ PS1='\[\e[1;38;5;141m\]╭─ \[\e[1;38;5;75m\]\u\[\e[0m\] \[\e[38;5;245m\]at\[\e[0m\] \[\e[1;38;5;114m\]\h\[\e[0m\] \[\e[38;5;245m\]in\[\e[0m\] \[\e[1;38;5;221m\]\w\[\e[0m\]\n\[\e[1;38;5;141m\]╰─ ❯ \[\e[0m\]' EOF # -------------------------------------------------------- # Verificar que la configuración se haya agregado # -------------------------------------------------------- if grep -q "$MARKER" "$BASHRC"; then echo -e "${GREEN}[SUCCESS]${RESET} Prompt personalizado instalado correctamente." else echo -e "${RED}[ERROR]${RESET} No se pudo instalar el prompt." exit 1 fi fi # ------------------------------------------------------------ # Instrucciones finales # ------------------------------------------------------------ echo "" echo -e "${PURPLE}╭──────────────────────────────────────────╮${RESET}" echo -e "${PURPLE}│ INSTALACIÓN FINALIZADA │${RESET}" echo -e "${PURPLE}╰──────────────────────────────────────────╯${RESET}" echo "" echo -e "${CYAN}[INFO]${RESET} Para aplicar los cambios inmediatamente ejecuta:" echo "" echo -e " ${GREEN}source ~/.bashrc${RESET}" echo "" echo -e "${GRAY}También puedes cerrar y abrir nuevamente tu terminal.${RESET}" echo ""
